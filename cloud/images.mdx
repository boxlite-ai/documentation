A box on BoxLite Cloud can start from an image in a public registry, or in a private one you log in to. The first box from an image records it in your organization's **image catalog**, and from then on the catalog decides which version a tag boots:

<CodeGroup>

```python Python
# first_image.py — start a box from a public image, then read the image back from the catalog
# Run: python first_image.py
import asyncio
import os
import time

from boxlite import ApiKeyCredential, Boxlite, BoxliteRestOptions, BoxOptions, NotFoundError

IMAGE = "docker.io/library/alpine:3.20"
NAME = "docker.io/library/alpine"

api_key = os.environ.get("BOXLITE_API_KEY")
if not api_key:
    raise SystemExit("Set BOXLITE_API_KEY to your blk_live_... key before running this.")


async def recorded(rt, name: str, tag: str):
    """The image and the version `tag` names, once the catalog has recorded it.

    The runner reports a started box's image on its next sync, about ten seconds
    later, so poll for up to 30 seconds while the box is still running.
    """
    for _ in range(30):
        try:
            detail = await rt.images.get(name)
            version = next((v for v in detail.versions if tag in v.tags), None)
            if version is not None:
                return detail, version
        except NotFoundError:
            pass
        await asyncio.sleep(1)
    raise TimeoutError(f"{name}:{tag} was not recorded within 30 seconds")


async def main() -> None:
    rt = Boxlite.rest(
        BoxliteRestOptions(
            url=os.environ.get("BOXLITE_REST_URL", "https://api.boxlite.ai"),
            credential=ApiKeyCredential(api_key),
        )
    )
    box_id = None
    try:
        # The first box from an image pulls it and records it in your catalog.
        box = await rt.create(BoxOptions(image=IMAGE), name=f"first-image-{int(time.time())}")
        box_id = box.id
        await box.start()

        execution = await box.exec("cat", args=["/etc/alpine-release"])
        release = "".join([line async for line in execution.stdout()]).strip()
        await execution.wait()
        print(f"box {box.id} runs Alpine {release}")

        # The catalog files the image under its full name, with the version it booted.
        detail, version = await recorded(rt, NAME, "3.20")
        print(f"{detail.name}  {version.digest}  tags={version.tags}")
    except Exception as exc:
        print(f"failed: {type(exc).__name__}: {exc}")
    finally:
        if box_id:
            await rt.remove(box_id, force=True)


if __name__ == "__main__":
    asyncio.run(main())
```

```typescript Node.js
// first_image.ts — start a box from a public image, then read the image back from the catalog
// Run: npx tsx first_image.ts
import { ApiKeyCredential, BoxliteRestOptions, JsBoxlite, errorCode } from "@boxlite-ai/boxlite";

const IMAGE = "docker.io/library/alpine:3.20";
const NAME = "docker.io/library/alpine";

type Runtime = ReturnType<typeof JsBoxlite.rest>;

// The image and the version `tag` names, once the catalog has recorded it.
// The runner reports a started box's image on its next sync, about ten seconds
// later, so poll for up to 30 seconds while the box is still running.
async function recorded(rt: Runtime, name: string, tag: string) {
  for (let attempt = 0; attempt < 30; attempt++) {
    try {
      const detail = await rt.images.get(name);
      const version = detail.versions.find((v) => v.tags.includes(tag));
      if (version) return { detail, version };
    } catch (err) {
      if (errorCode(err) !== "not_found") throw err;
    }
    await new Promise((resolve) => setTimeout(resolve, 1000));
  }
  throw new Error(`${name}:${tag} was not recorded within 30 seconds`);
}

async function main(): Promise<void> {
  const apiKey = process.env.BOXLITE_API_KEY;
  if (!apiKey) {
    throw new Error("Set BOXLITE_API_KEY to your blk_live_... key before running this.");
  }
  const rt = JsBoxlite.rest(
    new BoxliteRestOptions({
      url: process.env.BOXLITE_REST_URL ?? "https://api.boxlite.ai",
      credential: new ApiKeyCredential(apiKey),
    }),
  );

  let boxId: string | null = null;
  try {
    // The first box from an image pulls it and records it in your catalog.
    const box = await rt.create({ image: IMAGE }, `first-image-${Math.floor(Date.now() / 1000)}`);
    boxId = box.id;
    await box.start();

    const execution = await box.exec("cat", ["/etc/alpine-release"]);
    const stdout = await execution.stdout();
    let release = "";
    for (let line = await stdout.next(); line !== null; line = await stdout.next()) {
      release += line;
    }
    await execution.wait();
    console.log(`box ${box.id} runs Alpine ${release.trim()}`);

    // The catalog files the image under its full name, with the version it booted.
    const { detail, version } = await recorded(rt, NAME, "3.20");
    console.log(`${detail.name}  ${version.digest}  tags=${version.tags}`);
  } catch (err) {
    console.error("failed:", err);
  } finally {
    if (boxId !== null) {
      await rt.remove(boxId, true);
    }
    rt.close();
  }
}

main();
```

```go Go
// first_image — start a box from a public image, then read the image back from the catalog
// Run: go run .
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"slices"
	"strings"
	"time"

	boxlite "github.com/boxlite-ai/boxlite/sdks/go"
)

const (
	image = "docker.io/library/alpine:3.20"
	name  = "docker.io/library/alpine"
)

// recorded returns the image and the version tag names, once the catalog has
// recorded it. The runner reports a started box's image on its next sync, about
// ten seconds later, so poll for up to 30 seconds while the box is still running.
func recorded(ctx context.Context, images *boxlite.Images, name, tag string) (*boxlite.ImageDetail, *boxlite.ImageVersion, error) {
	for range 30 {
		detail, err := images.Get(ctx, name)
		if err != nil && !boxlite.IsNotFound(err) {
			return nil, nil, err
		}
		if detail != nil {
			for i := range detail.Versions {
				if slices.Contains(detail.Versions[i].Tags, tag) {
					return detail, &detail.Versions[i], nil
				}
			}
		}
		time.Sleep(time.Second)
	}
	return nil, nil, fmt.Errorf("%s:%s was not recorded within 30 seconds", name, tag)
}

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	cred, ok := boxlite.ApiKeyCredentialFromEnv()
	if !ok {
		return fmt.Errorf("set BOXLITE_API_KEY to your blk_live_... key before running this")
	}
	url := os.Getenv("BOXLITE_REST_URL")
	if url == "" {
		url = "https://api.boxlite.ai"
	}
	rt, err := boxlite.NewRest(boxlite.BoxliteRestOptions{URL: url, Credential: cred})
	if err != nil {
		return err
	}
	defer rt.Close()
	ctx := context.Background()
	images, err := rt.Images()
	if err != nil {
		return err
	}
	defer images.Close()

	// The first box from an image pulls it and records it in your catalog.
	box, err := rt.Create(ctx, image, boxlite.WithName(fmt.Sprintf("first-image-%d", time.Now().Unix())))
	if err != nil {
		return err
	}
	defer rt.ForceRemove(ctx, box.ID())
	defer box.Close()
	if err := box.Start(ctx); err != nil {
		return err
	}
	result, err := box.Exec(ctx, "cat", "/etc/alpine-release")
	if err != nil {
		return err
	}
	fmt.Printf("box %s runs Alpine %s\n", box.ID(), strings.TrimSpace(result.Stdout))

	// The catalog files the image under its full name, with the version it booted.
	detail, version, err := recorded(ctx, images, name, "3.20")
	if err != nil {
		return err
	}
	fmt.Printf("%s  %s  tags=%v\n", detail.Name, version.Digest, version.Tags)
	return nil
}
```

```bash REST
#!/usr/bin/env bash
# first_image.sh — start a box from a public image, then read the image back from the catalog
set -euo pipefail
: "${BOXLITE_API_KEY:?Set BOXLITE_API_KEY to your blk_live_... key before running this.}"

BOXLITE_REST_URL="${BOXLITE_REST_URL:-https://api.boxlite.ai}"
AUTH=(-H "Authorization: Bearer ${BOXLITE_API_KEY}")
NAME_PATH="docker.io%2Flibrary%2Falpine"   # the image name, URL-encoded as one path segment

# Print the digest of the version tag $1 names, once the catalog has recorded it.
# The runner reports a started box's image on its next sync, about ten seconds
# later, so poll for up to 30 seconds while the box is still running.
recorded() {
  local digest
  for _ in $(seq 30); do
    digest=$(curl -fsS "${BOXLITE_REST_URL}/v1/images/${NAME_PATH}" "${AUTH[@]}" 2>/dev/null \
      | jq -r --arg tag "$1" '.versions[] | select(.tags | index($tag)) | .digest' || true)
    if [ -n "${digest}" ]; then echo "${digest}"; return 0; fi
    sleep 1
  done
  echo "the version tagged $1 was not recorded within 30 seconds" >&2
  return 1
}

# The first box from an image pulls it and records it in your catalog.
BOX_ID=$(curl -fsS -X POST "${BOXLITE_REST_URL}/v1/boxes" "${AUTH[@]}" \
  -H 'Content-Type: application/json' \
  -d "{\"image\":\"docker.io/library/alpine:3.20\",\"name\":\"first-image-$(date +%s)\"}" | jq -r .box_id)
trap 'curl -fsS -X DELETE "${BOXLITE_REST_URL}/v1/boxes/${BOX_ID}?force=true" "${AUTH[@]}"' EXIT
echo "box ${BOX_ID}"

# The catalog files the image under its full name, with the version it booted.
echo "docker.io/library/alpine  $(recorded 3.20)"
```

## Prerequisites

- An API key from the console, exported as `BOXLITE_API_KEY`, and the REST URL exported as `BOXLITE_REST_URL`. See [API keys](/cloud/api-keys) and [Quickstart](/cloud/quickstart).
- An SDK:
  - Python: `pip install boxlite`.
  - Node.js: `npm install @boxlite-ai/boxlite`; run the TypeScript examples with `npx tsx`.
  - Go: `go get github.com/boxlite-ai/boxlite/sdks/go`, then `go run github.com/boxlite-ai/boxlite/sdks/go/cmd/setup` once to fetch the native library.
  - REST: `curl` and `jq`.

## What is the image catalog?

The catalog is your organization's record of the images its boxes have booted. Each entry is an **image name** — the registry host and repository, such as `docker.io/library/alpine` — and holds:

- **Versions.** Each version is one build, named by its manifest digest. The catalog keeps the reference it was first pulled as, its declared size, when it was recorded, and when a box last started from it.
- **Tags.** A tag points at one version. The first box created with a tag records where the tag pointed at that moment.
- **History.** Every time a tag is recorded or moved, the catalog appends an event naming both versions by digest.

```text
docker.io/library/alpine
├── sha256:45e0…  tags: 3.20, stable
└── sha256:5cd7…  tags: 3.19
```

Once a tag is recorded, a box created with it boots the recorded version — the catalog hands the runner `docker.io/library/alpine@sha256:45e0…`, not the tag. The registry is not asked again, so a tag that moves upstream does not move in your catalog. You decide when a tag moves: see [Pin a tag and roll it back](#pin-a-tag-and-roll-it-back).

The catalog records an image when the runner reports that the box started, about ten seconds after `create()` returns. Keep the box running until then: a box removed before the report leaves nothing recorded. That is why the examples that read the catalog poll `get()` for up to 30 seconds.

## Start a box from a public image

Pass a full image reference, such as `docker.io/library/alpine:3.20`, as the box's image. BoxLite Cloud admits public images from a set of registries that includes `docker.io`, `quay.io`, `gcr.io`, and `public.ecr.aws`. An image on a registry outside the set is refused with `Image registry '<host>' is not allowed. Allowed registries: …`, and the message lists the whole set.

<Warning>
**`base`, `python`, and `node` are not Docker Hub names on Cloud.** They select the console's images, so `image="python"` boots BoxLite's Python image, not `docker.io/library/python`. Write the full name, such as `docker.io/library/python:3.12-slim`, to use Docker Hub's. A short name like `alpine` is read as `docker.io/library/alpine:latest`.
</Warning>

A reference the catalog cannot answer yet is a **cold pull**. Your organization may start only a limited number of cold pulls in a short window; past it, `create()` fails with a resource-exhausted error whose message says how many seconds are left. Boxes from images the catalog already holds do not count.

When a pull fails, `create()` raises an invalid-argument error that says why:

| Message contains | Cause |
|---|---|
| `has no such image` | The repository or tag does not exist |
| `refused the pull` | The image is private, or does not exist, and no registry login covers it |
| `is not allowed` | The registry is not one Cloud admits |

## Read the catalog

`images.list()` returns one row per tag at its version, then one row per untagged version as `name@digest` with the tag `<none>`. The row's `id` is the version's digest. `images.get(name)` returns the image with every version and its tag history. It takes the name without a tag or digest — `docker.io/library/alpine`, not `alpine:3.20`.

| `get()` field | Python / Go | Node.js | Meaning |
|---|---|---|---|
| Name | `name` / `Name` | `name` | Registry host and repository |
| Tags | `tags` / `Tags` | `tags` | Every tag the image holds |
| Versions | `versions` / `Versions` | `versions` | Newest first |
| History | `history` / `History` | `history` | The latest 100 tag events, newest first |

| Version field | Python / Go | Node.js | Meaning |
|---|---|---|---|
| Digest | `digest` / `Digest` | `digest` | Manifest digest, such as `sha256:45e0…` |
| Tags | `tags` / `Tags` | `tags` | Tags that point at this version now |
| Source | `source_ref` / `SourceRef` | `sourceRef` | The reference the version was first pulled as |
| Size | `size_bytes` / `SizeBytes` | `sizeBytes` | Sum of the layer sizes the manifest declares; empty when unknown |
| Recorded | `recorded_at` / `RecordedAt` | `recordedAt` | When the catalog recorded it |
| Last used | `last_used_at` / `LastUsedAt` | `lastUsedAt` | When a box last started from it; empty when unknown |
| Missing upstream | `upstream_missing_at` / `UpstreamMissingAt` | `upstreamMissingAt` | When a pull found it gone from its registry; empty otherwise |

| History field | Python / Go | Node.js | Meaning |
|---|---|---|---|
| Tag | `tag` / `Tag` | `tag` | The tag that moved |
| From | `from_digest` / `FromDigest` | `fromDigest` | The version it pointed at before; empty when the tag was new |
| To | `to_digest` / `ToDigest` | `toDigest` | The version it points at since |
| Actor | `actor` / `Actor` | `actor` | Who moved it; empty when a pull recorded it |
| Moved | `moved_at` / `MovedAt` | `movedAt` | When |

Python returns times as RFC 3339 strings, Node.js as RFC 3339 strings, and Go as `time.Time`.

## Use a private image

To pull a private image, add a **registry login** for its registry. The password goes up once into a secret store, and no call ever returns it:

<CodeGroup>

```python Python
# private_image.py — add a registry login, start a box from a private image, then clean up
# Run: python private_image.py
import asyncio
import os
import time

from boxlite import ApiKeyCredential, Boxlite, BoxliteRestOptions, BoxOptions

PRIVATE_IMAGE = os.environ.get("PRIVATE_IMAGE", "ghcr.io/acme/app:1.0")
REGISTRY_HOST = PRIVATE_IMAGE.split("/", 1)[0]
# Whole path segments ending in "/": this login covers ghcr.io/acme/app and nothing else.
REPOSITORY_PREFIX = os.environ.get("REPOSITORY_PREFIX", "acme/app/")

api_key = os.environ.get("BOXLITE_API_KEY")
if not api_key:
    raise SystemExit("Set BOXLITE_API_KEY to your blk_live_... key before running this.")


async def main() -> None:
    rt = Boxlite.rest(
        BoxliteRestOptions(
            url=os.environ.get("BOXLITE_REST_URL", "https://api.boxlite.ai"),
            credential=ApiKeyCredential(api_key),
        )
    )
    login = None
    box_id = None
    try:
        # The password goes up once and is never returned.
        login = await rt.registries.create(
            registry_host=REGISTRY_HOST,
            username=os.environ["REGISTRY_USERNAME"],
            password=os.environ["REGISTRY_PASSWORD"],
            repository_prefix=REPOSITORY_PREFIX,
        )
        print(f"login {login.id} covers {login.registry_host}/{login.repository_prefix}*")

        # Boxes from any image the login covers now pull with it.
        box = await rt.create(BoxOptions(image=PRIVATE_IMAGE), name=f"private-{int(time.time())}")
        box_id = box.id
        await box.start()
        execution = await box.exec("cat", args=["/etc/os-release"])
        output = "".join([chunk async for chunk in execution.stdout()])
        first_line = output.splitlines()[0]
        await execution.wait()
        print(f"box {box.id} from {PRIVATE_IMAGE}: {first_line}")
    except Exception as exc:
        print(f"failed: {type(exc).__name__}: {exc}")
    finally:
        if box_id:
            await rt.remove(box_id, force=True)
        # A login is refused removal while a box still pulls through it, so remove boxes first.
        if login is not None:
            await rt.registries.remove(login.id)


if __name__ == "__main__":
    asyncio.run(main())
```

```go Go
// private_image — add a registry login, start a box from a private image, then clean up
// Run: go run .
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"strings"
	"time"

	boxlite "github.com/boxlite-ai/boxlite/sdks/go"
)

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	cred, ok := boxlite.ApiKeyCredentialFromEnv()
	if !ok {
		return fmt.Errorf("set BOXLITE_API_KEY to your blk_live_... key before running this")
	}
	url := os.Getenv("BOXLITE_REST_URL")
	if url == "" {
		url = "https://api.boxlite.ai"
	}
	image := os.Getenv("PRIVATE_IMAGE")
	if image == "" {
		image = "ghcr.io/acme/app:1.0"
	}
	// Whole path segments ending in "/": this login covers ghcr.io/acme/app and nothing else.
	prefix := os.Getenv("REPOSITORY_PREFIX")
	if prefix == "" {
		prefix = "acme/app/"
	}
	rt, err := boxlite.NewRest(boxlite.BoxliteRestOptions{URL: url, Credential: cred})
	if err != nil {
		return err
	}
	defer rt.Close()
	ctx := context.Background()
	registries, err := rt.Registries()
	if err != nil {
		return err
	}
	defer registries.Close()

	// The password goes up once and is never returned.
	login, err := registries.Create(ctx, boxlite.NewRegistryCredential{
		RegistryHost:     strings.SplitN(image, "/", 2)[0],
		RepositoryPrefix: prefix,
		Username:         os.Getenv("REGISTRY_USERNAME"),
		Password:         os.Getenv("REGISTRY_PASSWORD"),
	})
	if err != nil {
		return err
	}
	// A login is refused removal while a box still pulls through it, so this
	// runs after the box removal deferred below.
	defer registries.Remove(ctx, login.ID)
	fmt.Printf("login %s covers %s/%s*\n", login.ID, login.RegistryHost, login.RepositoryPrefix)

	// Boxes from any image the login covers now pull with it.
	box, err := rt.Create(ctx, image, boxlite.WithName(fmt.Sprintf("private-%d", time.Now().Unix())))
	if err != nil {
		return err
	}
	defer rt.ForceRemove(ctx, box.ID())
	defer box.Close()
	if err := box.Start(ctx); err != nil {
		return err
	}
	result, err := box.Exec(ctx, "cat", "/etc/os-release")
	if err != nil {
		return err
	}
	firstLine, _, _ := strings.Cut(result.Stdout, "\n")
	fmt.Printf("box %s from %s: %s\n", box.ID(), image, firstLine)
	return nil
}
```

```bash REST
#!/usr/bin/env bash
# private_image.sh — add a registry login, start a box from a private image, then clean up
set -euo pipefail
: "${BOXLITE_API_KEY:?Set BOXLITE_API_KEY to your blk_live_... key before running this.}"
: "${REGISTRY_USERNAME:?Set REGISTRY_USERNAME}" "${REGISTRY_PASSWORD:?Set REGISTRY_PASSWORD}"

AUTH=(-H "Authorization: Bearer ${BOXLITE_API_KEY}")
PRIVATE_IMAGE="${PRIVATE_IMAGE:-ghcr.io/acme/app:1.0}"

LOGIN_ID=$(jq -n --arg host "${PRIVATE_IMAGE%%/*}" --arg prefix "${REPOSITORY_PREFIX}" \
    --arg user "${REGISTRY_USERNAME}" --arg pass "${REGISTRY_PASSWORD}" \
    '{registry_host: $host, repository_prefix: $prefix, username: $user, password: $pass}' \
  | curl -fsS -X POST "${BOXLITE_REST_URL}/v1/registries" "${AUTH[@]}" \
      -H 'Content-Type: application/json' -d @- | jq -r .id)
cleanup() {
  if [ -n "${BOX_ID}" ]; then
    curl -fsS -X DELETE "${BOXLITE_REST_URL}/v1/boxes/${BOX_ID}?force=true" "${AUTH[@]}"
  fi
  # A login is refused removal while a box still pulls through it, so remove boxes first.
  curl -fsS -X DELETE "${BOXLITE_REST_URL}/v1/registries/${LOGIN_ID}" "${AUTH[@]}"
```

A login covers every repository under its `repository_prefix` on its registry. The example scopes it to one repository with `acme/app/`. Set the prefix to whole path segments ending in `/`: `acme/` covers `ghcr.io/acme/app` but not `ghcr.io/acme-other/app`. Leave it out to cover the whole registry. When two logins cover an image, the one with the longer prefix wins.

| Rule | Detail |
|---|---|
| Registries that take a login | `ghcr.io`, `docker.io`, `quay.io`, `gcr.io`, `us.gcr.io`, `eu.gcr.io`, `asia.gcr.io` |
| One login per registry and prefix | A second one raises an already-exists error; remove the first to replace it |
| Removal | Refused, naming the boxes, while a box still pulls through the login. Remove those boxes first |
| Listing | `registries.list()` returns every login, oldest first, without passwords |

A login also admits public images under its prefix, on a registry that takes logins, even if that registry is outside the set Cloud admits for anonymous pulls.

## Pin a tag and roll it back

`images.move_tag(name, tag, digest)` points a tag at one of the image's recorded versions, and creates the tag if it does not exist. The next box created with that tag boots that version. Boxes already created keep the version they booted, and nothing is pulled. Each move is added to the image's history.

This example records two versions, moves `stable` to the newer one, then rolls it back:

<CodeGroup>

```python Python
import os
import time

from boxlite import ApiKeyCredential, Boxlite, BoxliteRestOptions, BoxOptions, NotFoundError

TAG = "stable"

                moved_from = event.from_digest or "(new)"
```

```go Go
func boot(ctx context.Context, rt *boxlite.Runtime, image string) (string, error) {
	box, err := rt.Create(ctx, image, boxlite.WithName(fmt.Sprintf("pin-a-tag-%d", time.Now().UnixMilli())))
	if err != nil {
		return "", err
	}
	defer box.Close()

		return box.ID(), err
	}
	result, err := box.Exec(ctx, "cat", "/etc/alpine-release")
	if err != nil {
		return box.ID(), err
	}
	fmt.Printf("%s boots Alpine %s\n", image, strings.TrimSpace(result.Stdout))

// recorded it. The runner reports a started box's image on its next sync, about
// ten seconds later, so poll for up to 30 seconds while the box is still running.
func recorded(ctx context.Context, images *boxlite.Images, name, tag string) (*boxlite.ImageDetail, *boxlite.ImageVersion, error) {

		}

			for i := range detail.Versions {
				if slices.Contains(detail.Versions[i].Tags, tag) {
					return detail, &detail.Versions[i], nil
				}
			}
		}

	}
	return nil, nil, fmt.Errorf("%s:%s was not recorded within 30 seconds", name, tag)
}

	if err := run(); err != nil {
		log.Fatal(err)
	}
}

	cred, ok := boxlite.ApiKeyCredentialFromEnv()
	if !ok {
		return fmt.Errorf("set BOXLITE_API_KEY to your blk_live_... key before running this")

	if url == "" {
		url = "https://api.boxlite.ai"

		return err

	if err != nil {
		return err
	}
	defer images.Close()

	var boxes []string
	defer func() {
		for _, id := range boxes {
			rt.ForceRemove(ctx, id)
		}
	}()
	run := func(image string) error {
		id, err := boot(ctx, rt, image)
		if id != "" {
			boxes = append(boxes, id)
		}
		return err

	// Two tags of one repository record two versions of one image.

	if _, err := images.MoveTag(ctx, name, tag, newer.Digest); err != nil {
		return err
```

The digest must belong to a version of that image that the catalog holds. Any other digest is refused with `Digest '<digest>' is not a version of image '<name>'`.

### Pick up where a tag points upstream

A pull never moves a recorded tag. To move a tag to a newer upstream build, either:

- Boot a box from a reference the catalog has not recorded, such as a new tag or `name@digest`. That records the build as a version, and you can then move your tag to it.
- Remove the image from the catalog and create a box with the tag again. The next box resolves the tag against the registry afresh. Removing an image drops its other tags and its history too.

### When a version disappears upstream

A registry can delete a build that one of your tags still points at. Runners that already hold the build keep starting it. On a runner that does not, `create()` fails with a message such as:

```text
Failed to pull image 'docker.io/acme/app@sha256:…': docker.io no longer has it (MANIFEST_UNKNOWN). Boxes on runners that already hold it still start. Move the tag 'stable' to a version its registry still has, or remove the image from the catalog and use the reference again.
```

The catalog also marks the version: its `upstream_missing_at` field holds when a pull first found it gone. Move the tag to a version the registry still has, as the message says.

## Keep the catalog under its limit

An organization holds a limited number of images, 20 by default. `images.usage()` reports how many you hold, your limit, and the sum of their declared sizes. Once you reach the limit, a box from an image that is new to the catalog is refused; boxes from images you already hold still start.

`images.remove(name)` removes an image from the catalog. It is refused, naming the boxes, while any box can still boot from it — including a stopped one. Runners keep the bytes they cached, and using the image again records it as a new entry.

```go Go
	if err != nil {
		return err
	}
	defer images.Close()

	box, err := rt.Create(ctx, image, boxlite.WithName(fmt.Sprintf("catalog-%d", time.Now().Unix())))
	if err != nil {
		return err
	}
	boxID := box.ID()
	box.Close()
	defer func() {
		if boxID != "" {
			rt.ForceRemove(ctx, boxID)
		}
	}()
	if _, _, err := recorded(ctx, images, name, "3.20"); err != nil {
		return err
	}

	usage, err := images.Usage(ctx)
	if err != nil {
		return err
	}
	fmt.Printf("holding %d of %d images, %d bytes declared\n", usage.Count, usage.Limit, usage.KnownBytes)
	held, err := images.List(ctx)
	if err != nil {
		return err
	}
	for _, info := range held {
		fmt.Printf("%s  %s\n", info.Reference, info.ID)
	}

	// An image a box can still boot from cannot be removed.
	if err := images.Remove(ctx, name); boxlite.IsInvalidState(err) {
		fmt.Printf("refused: %v\n", strings.TrimSpace(err.Error()))
	} else if err != nil {
		return err
	}

	if err := rt.ForceRemove(ctx, boxID); err != nil {
		return err
	}
	boxID = ""
	if err := images.Remove(ctx, name); err != nil {
		return err
	}
	if usage, err = images.Usage(ctx); err != nil {
		return err
	}
	fmt.Printf("removed %s; holding %d images\n", name, usage.Count)
	return nil
}
```

```bash REST
#!/usr/bin/env bash
# manage_catalog.sh — read usage, list the catalog, and remove an image once no box needs it
set -euo pipefail
: "${BOXLITE_API_KEY:?Set BOXLITE_API_KEY to your blk_live_... key before running this.}"

AUTH=(-H "Authorization: Bearer ${BOXLITE_API_KEY}")
NAME_PATH="docker.io%2Flibrary%2Falpine"

# Print the digest of the version tag $1 names, once the catalog has recorded it.
# The runner reports a started box's image on its next sync, about ten seconds
# later, so poll for up to 30 seconds while the box is still running.
recorded() {
  local digest
  for _ in $(seq 30); do
    digest=$(curl -fsS "${BOXLITE_REST_URL}/v1/images/${NAME_PATH}" "${AUTH[@]}" 2>/dev/null \
      | jq -r --arg tag "$1" '.versions[] | select(.tags | index($tag)) | .digest' || true)
    if [ -n "${digest}" ]; then echo "${digest}"; return 0; fi
    sleep 1
  done
  echo "the version tagged $1 was not recorded within 30 seconds" >&2
  return 1
}

BOX_ID=$(curl -fsS -X POST "${BOXLITE_REST_URL}/v1/boxes" "${AUTH[@]}" \
  -H 'Content-Type: application/json' \
  -d "{\"image\":\"docker.io/library/alpine:3.20\",\"name\":\"catalog-$(date +%s)\"}" | jq -r .box_id)
recorded 3.20 >/dev/null

curl -fsS "${BOXLITE_REST_URL}/v1/images/usage" "${AUTH[@]}" \
  | jq -r '"holding \(.count) of \(.limit) images, \(.known_bytes) bytes declared"'
curl -fsS "${BOXLITE_REST_URL}/v1/images" "${AUTH[@]}" | jq -r '.images[] | "\(.reference)  \(.id)"'

# An image a box can still boot from cannot be removed: 409, naming the box.
curl -sS -X DELETE "${BOXLITE_REST_URL}/v1/images/${NAME_PATH}" "${AUTH[@]}" | jq -r '"refused: \(.message)"'

curl -fsS -X DELETE "${BOXLITE_REST_URL}/v1/boxes/${BOX_ID}?force=true" "${AUTH[@]}"
curl -fsS -X DELETE "${BOXLITE_REST_URL}/v1/images/${NAME_PATH}" "${AUTH[@]}"
curl -fsS "${BOXLITE_REST_URL}/v1/images/usage" "${AUTH[@]}" | jq -r '"removed; holding \(.count) images"'
```

## The console's images

The console's **Base**, **Python**, and **Node.js** images — `base`, `python`, and `node` from code — are provided by BoxLite and shared by every organization. They are not part of your catalog: `list()` and `usage()` leave them out, and they do not count against your limit. `get()` on one of them answers with `curated` set and no versions. Removing one, or moving a tag of one, is refused.

## API key permissions

Each call needs a permission on the API key:

| Call | Permission |
|---|---|
| `images.list()`, `images.get()`, `images.usage()` | `read:images` |
| `images.move_tag()` | `write:images` |
| `images.remove()` | `delete:images` |
| `registries.list()` | `read:registries` |
| `registries.create()` | `write:registries` |
| `registries.remove()` | `delete:registries` |

Creating a box records its image without any image permission. A key created before these permissions existed does not have them, and the call fails with a forbidden error (`403`). Create a new key in the console.

## Errors

| Situation | Python | Node.js `errorCode(err)` | Go |
|---|---|---|---|
| Pull failed, registry not allowed, bad reference, digest not a version, image limit reached | `InvalidArgumentError` | `invalid_argument` | `ErrInvalidArgument` |
| Image or login not found | `NotFoundError` | `not_found` | `ErrNotFound` |
| A login for that registry and prefix exists | `AlreadyExistsError` | `already_exists` | `ErrAlreadyExists` |
| Image or login still in use by a box | `InvalidStateError` | `invalid_state` | `ErrInvalidState` |
| Too many cold pulls in the window | `ResourceExhaustedError` | `resource_exhausted` | `ErrResourceExhausted` |
| The key lacks the permission | `ConfigError` | `config_error` | `ErrConfig` |
| `pull()`, which a Cloud box does at creation instead | `UnsupportedError` | `unsupported` | `ErrUnsupported` |

Over REST, the first six rows are `400`, `404`, `409` with the code `already_exists`, `409`, `429`, and `403`. There is no pull route.
